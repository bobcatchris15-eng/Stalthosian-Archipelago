cmake_minimum_required(VERSION 3.25)
project(StalthosianArchipelago LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/scenarios.cpp")
    add_executable(StalthosScenarios tests/scenarios.cpp)
    target_include_directories(StalthosScenarios PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME domain-scenarios COMMAND StalthosScenarios)
endif()
add_executable(StalthosPersistence tests/persistence.cpp)
target_include_directories(StalthosPersistence PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(StalthosPersistence PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(StalthosPersistence PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME domain-persistence COMMAND StalthosPersistence)
add_executable(StalthosLandscape tests/reference_landscape.cpp)
if(MSVC)
    target_compile_options(StalthosLandscape PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(StalthosLandscape PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME reference-landscape COMMAND StalthosLandscape)
add_executable(StalthosLifecycle tests/lifecycle.cpp)
if(MSVC)
    target_compile_options(StalthosLifecycle PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(StalthosLifecycle PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME domain-lifecycle COMMAND StalthosLifecycle)
foreach(MODULE animation_blend needs relationships activation_plan environment_palette habitat_dressing shore_mesh locomotion_clip locomotion_pace animation_transition terrain_texture rendered_ground_surface body_terrain_support foot_support_plane foot_terrain_diagnostic resource_ledger resource_presentation woodland_stands camera_framing hero_view_controller fauna_population world_environment archipelago_world fauna_resources terrain_tiles watershed_terrain)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${MODULE}.cpp")
        add_executable(Stalthos_${MODULE} "tests/${MODULE}.cpp")
        target_include_directories(Stalthos_${MODULE} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_features(Stalthos_${MODULE} PRIVATE cxx_std_17)
        if(MSVC)
            target_compile_options(Stalthos_${MODULE} PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(Stalthos_${MODULE} PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
        add_test(NAME "domain-${MODULE}" COMMAND Stalthos_${MODULE})
    endif()
endforeach()
option(STALTHOS_BUILD_WICKED_HOST "Build the Windows Wicked game host" ON)
if(STALTHOS_BUILD_WICKED_HOST)
    if(NOT WIN32)
        message(FATAL_ERROR "Wicked host currently targets Windows; use -DSTALTHOS_BUILD_WICKED_HOST=OFF for domain tests")
    endif()
    if(NOT DEFINED WICKED_ENGINE_ROOT OR NOT DEFINED WICKED_RELEASE_CACHE)
        message(FATAL_ERROR "Run scripts/Prepare-Wicked.ps1 and scripts/Build.ps1")
    endif()
    file(GLOB ENGINE_OBJECTS "${WICKED_RELEASE_CACHE}/objects/*.obj")
    list(LENGTH ENGINE_OBJECTS ENGINE_OBJECT_COUNT)
    if(NOT ENGINE_OBJECT_COUNT EQUAL 80)
        message(FATAL_ERROR "Pinned Release cache requires 80 common engine objects")
    endif()
    set_source_files_properties(${ENGINE_OBJECTS} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
    file(READ "${WICKED_ENGINE_ROOT}/Editor/ModelImporter_GLTF.cpp" IMPORTER)
    string(REPLACE "#include \"stdafx.h\"" "#include \"WickedEngine.h\"" IMPORTER "${IMPORTER}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ModelImporter_GLTF.cpp" "${IMPORTER}")
    add_executable(Stalthos WIN32 engine/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/ModelImporter_GLTF.cpp" ${ENGINE_OBJECTS})
    target_include_directories(Stalthos PRIVATE "${WICKED_ENGINE_ROOT}/WickedEngine" "${WICKED_ENGINE_ROOT}/Editor" "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(Stalthos PRIVATE /utf-8 /arch:AVX /EHsc $<$<CONFIG:Release>:/O2>)
    target_compile_definitions(Stalthos PRIVATE WIN32 _WINDOWS WIN32=1 _HAS_EXCEPTIONS=0 NDEBUG
        _XM_AVX_INTRINSICS_ _XM_SSE4_INTRINSICS_ _XM_F16C_INTRINSICS_ _XM_FMA3_INTRINSICS_
        WICKED_CMAKE_BUILD UNICODE _UNICODE
        JPH_USE_AVX JPH_USE_SSE4_1 JPH_USE_SSE4_2 JPH_USE_LZCNT JPH_USE_TZCNT JPH_USE_F16C JPH_USE_FMADD)
    foreach(LIBRARY WickedEngine Jolt LUA Utility)
        target_link_libraries(Stalthos PRIVATE "${WICKED_RELEASE_CACHE}/libraries/${LIBRARY}.lib")
    endforeach()
    target_link_libraries(Stalthos PRIVATE bcrypt kernel32 user32 gdi32 winspool shell32 ole32 oleaut32 uuid comdlg32 advapi32)
    add_custom_command(TARGET Stalthos POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${WICKED_RELEASE_CACHE}/bin/dxcompiler.dll" "$<TARGET_FILE_DIR:Stalthos>/dxcompiler.dll"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${WICKED_ENGINE_ROOT}/LICENSE.txt" "$<TARGET_FILE_DIR:Stalthos>/WickedEngine-LICENSE.txt")
endif()
